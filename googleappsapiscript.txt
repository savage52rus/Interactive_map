const SECRET_TOKEN = "a3f8c9d2e1b74f6a8c5d3e9f1b2a7c4d8e6f0a3b5c7d9e1f2a4b6c8d0e2f4a6b"; // Обязательно поменяйте!
const SHEET_MARKERS = "Data";
const SHEET_ROADS   = "Roads";
const HEADERS_MARKERS = ["id", "type", "x", "y", "name", "desc", "color"];
const HEADERS_ROADS   = ["id", "weeks", "points"];

// ============================================================
// ВСПОМОГАТЕЛЬНЫЕ ФУНКЦИИ ДЛЯ ТАБЛИЦ
// ============================================================
function getOrCreateSheet(name, headers) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
  }
  const firstRow = sheet.getRange(1, 1, 1, Math.max(1, headers.length)).getValues()[0];
  const isEmpty = firstRow.every(v => v === "" || v === null);
  if (isEmpty) {
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  }
  return sheet;
}

function sheetToObjects(sheet) {
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  const headers = data[0];
  return data.slice(1)
    .filter(r => r[0] !== "" && r[0] !== null)
    .map(row => {
      const obj = {};
      headers.forEach((h, i) => { obj[h] = row[i]; });
      return obj;
    });
}

function findRowById(sheet, id) {
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return -1;
  const headers = data[0];
  const idCol = headers.indexOf("id");
  if (idCol < 0) return -1;
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][idCol]) === String(id)) return i + 1;
  }
  return -1;
}

// ============================================================
// GET — Чтение данных (с поддержкой JSONP)
// ============================================================
function doGet(e) {
  try {
    const kind = e && e.parameter ? e.parameter.kind : null;
    const callback = e && e.parameter ? e.parameter.callback : null;

    const markersSheet = getOrCreateSheet(SHEET_MARKERS, HEADERS_MARKERS);
    const roadsSheet   = getOrCreateSheet(SHEET_ROADS,   HEADERS_ROADS);

    const markersRaw = sheetToObjects(markersSheet).map(m => ({
      id: parseInt(m.id, 10) || 0,
      type: String(m.type || "poi"),
      x: Number(m.x) || 0,
      y: Number(m.y) || 0,
      name: String(m.name || ""),
      desc: String(m.desc || ""),
      color: String(m.color || "#c9a227")
    }));

    const roadsRaw = sheetToObjects(roadsSheet).map(r => {
      let pts = [];
      try { pts = JSON.parse(r.points || "[]"); } catch (err) { pts = []; }
      if (!Array.isArray(pts)) pts = [];
      pts = pts
        .map(p => ({ x: Number(p.x), y: Number(p.y) }))
        .filter(p => isFinite(p.x) && isFinite(p.y));
      return {
        id: parseInt(r.id, 10) || 0,
        weeks: parseInt(r.weeks, 10) || 1,
        points: pts
      };
    }).filter(r => r.points.length >= 2);

    let dataToReturn;
    if (kind === "markers") dataToReturn = markersRaw;
    else if (kind === "roads") dataToReturn = roadsRaw;
    else dataToReturn = { markers: markersRaw, roads: roadsRaw };

    const jsonString = JSON.stringify(dataToReturn);

    // Если передан callback (JSONP) — оборачиваем в функцию
    if (callback) {
      return ContentService
        .createTextOutput(callback + '(' + jsonString + ');')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }

    // Иначе отдаём обычный JSON
    return ContentService
      .createTextOutput(jsonString)
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    const callback = e && e.parameter ? e.parameter.callback : null;
    const errJson = JSON.stringify({ error: err.toString() });
    
    if (callback) {
      return ContentService
        .createTextOutput(callback + '(' + errJson + ');')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService
      .createTextOutput(errJson)
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ============================================================
// POST — Запись данных
// ============================================================
function doPost(e) {
  try {
    let payload;
    try {
      payload = JSON.parse(e.postData.contents);
    } catch (err) {
      throw new Error("Ошибка парсинга JSON.");
    }

    if (payload.token !== SECRET_TOKEN) {
      throw new Error("Неверный токен доступа.");
    }

    const kind = payload.kind || (payload.marker ? "marker" : "marker");

    if (kind === "marker") return handleMarkerPost(payload);
    if (kind === "road")   return handleRoadPost(payload);

    throw new Error("Неизвестный тип данных: " + kind);
  } catch (err) {
    // Для POST JSONP не используется, поэтому отдаем обычный JSON
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function handleMarkerPost(payload) {
  const sheet = getOrCreateSheet(SHEET_MARKERS, HEADERS_MARKERS);
  const headers = sheet.getRange(1, 1, 1, HEADERS_MARKERS.length).getValues()[0];

  if (payload.action === "upsert") {
    const m = payload.marker || payload.item || {};
    const row = headers.map(h => (m[h] !== undefined ? m[h] : ""));
    const foundRow = findRowById(sheet, m.id);
    
    if (foundRow > 0) {
      sheet.getRange(foundRow, 1, 1, headers.length).setValues([row]);
      return ContentService.createTextOutput(JSON.stringify({ status: 'ok', action: 'updated', id: m.id })).setMimeType(ContentService.MimeType.JSON);
    }
    sheet.appendRow(row);
    return ContentService.createTextOutput(JSON.stringify({ status: 'ok', action: 'created', id: m.id })).setMimeType(ContentService.MimeType.JSON);
  }

  if (payload.action === "delete") {
    const foundRow = findRowById(sheet, payload.id);
    if (foundRow > 0) {
      sheet.deleteRow(foundRow);
      return ContentService.createTextOutput(JSON.stringify({ status: 'ok', action: 'deleted', id: payload.id })).setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'Маркер не найден.' })).setMimeType(ContentService.MimeType.JSON);
  }

  throw new Error("Неизвестное действие для маркера: " + payload.action);
}

function handleRoadPost(payload) {
  const sheet = getOrCreateSheet(SHEET_ROADS, HEADERS_ROADS);
  const headers = sheet.getRange(1, 1, 1, HEADERS_ROADS.length).getValues()[0];

  if (payload.action === "upsert") {
    const r = payload.road || payload.item || {};
    const pointsStr = JSON.stringify(r.points || []);
    const row = headers.map(h => {
      if (h === "points") return pointsStr;
      return r[h] !== undefined ? r[h] : "";
    });
    
    const foundRow = findRowById(sheet, r.id);
    if (foundRow > 0) {
      sheet.getRange(foundRow, 1, 1, headers.length).setValues([row]);
      return ContentService.createTextOutput(JSON.stringify({ status: 'ok', action: 'updated', id: r.id })).setMimeType(ContentService.MimeType.JSON);
    }
    sheet.appendRow(row);
    return ContentService.createTextOutput(JSON.stringify({ status: 'ok', action: 'created', id: r.id })).setMimeType(ContentService.MimeType.JSON);
  }

  if (payload.action === "delete") {
    const foundRow = findRowById(sheet, payload.id);
    if (foundRow > 0) {
      sheet.deleteRow(foundRow);
      return ContentService.createTextOutput(JSON.stringify({ status: 'ok', action: 'deleted', id: payload.id })).setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'Дорога не найдена.' })).setMimeType(ContentService.MimeType.JSON);
  }

  throw new Error("Неизвестное действие для дороги: " + payload.action);
}
